@startuml
!pragma teoz true
skinparam style strictuml

' =========================
' ==== CLASES BASE ========
' =========================

class Persona {
  - nombre : String
  - apellido : String
  - email : String
  - telefono : int

  + crearPersona()
}

class Cliente {
  - idCliente : int

  + solicitarTurno(s : Servicio, p : Profesional, fechaHora : DateTime) : Turno
  + cancelarTurno(t : Turno) 
  + verMisTurnos() : List<Turno>
  + seleccionarTurno() : void
  + confirmarOperacion() : void
  + cancelarOperacion() : void
  + ingresarDatos(nombre:String, apellido:String, telefono:int, email:String) : Cliente
  + ingresarCodigo(codigo : String) 
}

class Profesional {
  - idProfesional : int
  - especialidad : String
  - disponibilidad : List<DateTime>
  
  + agregarProfesional(nombre:String, apellido:String, disponibilidad:DateTime, especialidad:String) : Profesional
  + eliminarProfesional(idProfesional : int)
  + modificarProfesional(nombre:String, apellido:String, disponibilidad:DateTime, especialidad:String) : Profesional
}

class Administrador {
  - idAdmin : int

  + confirmarOperacion() : void
  + cancelarOperacion() : void
  + ingresarDatos(nombre:String, apellido:String, telefono:int) : Administrador
  + ingresarCodigo(codigo : String) : void
  + seleccionarProfesionales(List<Profesional>)
  + actualizarDisponibilidad(fechaHora : DateTime)
}

class Servicio {
  - tipo : String
  - precio : float

  + agregarServicio(tipo:String, precio:float) : Servicio
  + eliminarServicio(tipo:String, precio:float) 
  + modificarServicio(tipo:String, precio:float) : Servicio
}

class Turno {
  - idTurno : int
  - fechaHora : DateTime
  - estado : String 
  - servicios : List<Servicio> 

  + cambiarEstado(Estado : String)
  + verificarAnticipacion() : Boolean
}

' =========================
' ==== CATALOGOS ==========
' =========================

class CatalogoClientes {
  - clientes : List<Cliente>
  + agregarCliente(c : Cliente)
  + buscarCliente(idCliente : int) : Cliente
}

class CatalogoProfesionales {
  - profesionales : List<Profesional>
  + agregarProfesional(p : Profesional)
  + eliminarProfesional(p : Profesional)
  + modificarProfesional(p : Profesional)
  + buscarProfesional(nombre : String) : Profesional
}

class CatalogoServicios {
  - servicios : List<Servicio>
  + agregarServicio(s : Servicio)
  + eliminarServicio(s : Servicio)
  + modificarServicio(s : Servicio)
  + buscarServicio(tipo : String) : Servicio
}

class CatalogoTurnos {
  - turnos : List<Turno>
  + agregarTurno(t : Turno)
  + eliminarTurno(t : Turno)
  + buscarTurno(idTurno : int) : Turno
}

class CatalogoAdministradores {
   - administradores : List<Administrador>
}

' =========================
' ==== CONTROLADOR ========
' =========================

class Controlador {
  - catalogoClientes : CatalogoClientes
  - catalogoProfesionales : CatalogoProfesionales
  - catalogoServicios : CatalogoServicios
  - catalogoTurnos : CatalogoTurnos
  - catalogoAdministradores : CatalogoAdministradores

  + agregarCliente(nombre:String, apellido:String, mail:String, telefono:int) : Cliente
  + eliminarCliente(idCliente : int) 

  + agregarProfesional(nombre:String, apellido:String, disponibilidad:DateTime, especialidad:String) : Profesional
  + eliminarProfesional(idProfesional : int)
  + modificarProfesional(disponibilidad:DateTime, especialidad:String) : Profesional

  + agregarServicio(tipo:String, precio:float) : Servicio
  + eliminarServicio(tipo:String, precio:float)
  + modificarServicio(tipo:String, precio:float) : Servicio

  + crearTurno(c:Cliente, p:Profesional, s:Servicio, fechaHora:DateTime) : Turno
  + cancelarTurno(idTurno : int) : Turno

  + validarDatos() : Boolean
  + enviarCodigo() : String
  + validarCodigo(codigo : String) : Boolean

  + buscarCliente(idCliente : int) : Cliente
  + buscarTurno(idTurno : int) : Turno
  + buscarServicio(tipo : String) : Servicio
  + buscarProfesional(idProfesional : int) : Profesional
}

' =========================
' ==== RELACIONES =========
' =========================

Persona <|-- Cliente
Persona <|-- Profesional
Persona <|-- Administrador

Cliente "1" -- "0..*" Turno : solicita >
Profesional "1" -- "0..*" Turno : atiende >
Turno "1" -- "1..*" Servicio : incluye >

CatalogoClientes "1" o-- "0..*" Cliente
CatalogoProfesionales "1" o-- "0..*" Profesional
CatalogoServicios "1" o-- "0..*" Servicio
CatalogoTurnos "1" o-- "0..*" Turno
CatalogoAdministradores "1" o-- "0..*" Administrador

Administrador --> Controlador
Controlador *-- CatalogoClientes
Controlador *-- CatalogoProfesionales
Controlador *-- CatalogoServicios
Controlador *-- CatalogoTurnos
Controlador *-- CatalogoAdministradores
@enduml
